{{- /* Markdown images get their intrinsic size so the layout (and anchor targets) do not shift while they load. */ -}} {{- $destination := .Destination -}} {{- $parsed := urls.Parse $destination -}} {{- $image := "" -}} {{- if not $parsed.IsAbs -}} {{- $image = resources.Get (strings.TrimPrefix "/" $parsed.Path) -}} {{- end -}} {{- $src := $destination -}} {{- with $image -}} {{- $src = .RelPermalink -}} {{- else -}} {{- if strings.HasPrefix $destination "/" -}} {{- $src = replaceRE `^/` `` $destination | relURL -}} {{- end -}} {{- end -}} {{- /* This comment removes trailing newlines. */ -}}
